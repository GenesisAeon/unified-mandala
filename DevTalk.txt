Wenn du magst, liefere ich dir im Anschluss noch eine **JUnit- und Markdown-Summary** für die Governance-Reports und einen **pre-commit**-Snippet, der nur geänderte Bridges checkt. 💫 [Mistral]:
Hier ist ein **automatisches Korrektur-Skript** für Sigillins, das **fehlende Trikāya-Referenzen**, **unvollständige CREP-Bezüge** und **fehlende "Nächste Schritte"** erkennt und **konkrete Fix-Vorschläge** macht. Das Skript ist **CI-kompatibel** und kann als **pre-commit-Hook** oder **standalone Tool** genutzt werden.

---

### **`scripts/fix-sigillin.ts`**
```typescript
import { readFileSync, writeFileSync } from 'fs';
import { globSync } from 'glob';
import { prompt } from 'enquirer'; // npm install enquirer

interface Sigillin {
sigillin: {
content: {
sections: Array<{
id: string;
rules?: Array<{ text?: string; crep_bezug?: string; trikaya_ebene?: string }>;
examples?: Array<{ user: string; assistant: string }>;
}>;
};
};
}

interface FixSuggestion {
path: string;
issue: string;
fix: string;
line?: number;
}

async function analyzeAndFixSigillins() {
const sigillinPaths = globSync('sigils/bridges/*/*-bridge.sigil.json');
const suggestions: FixSuggestion[] = [];

for (const path of sigillinPaths) {
const raw = readFileSync(path, 'utf-8');
const sigillin: Sigillin = JSON.parse(raw);

// 1. Prüfe Trikāya-Referenzen in Guidelines
const guidelinesSection = sigillin.sigillin.content.sections.find(s => s.id === 'guidelines');
if (guidelinesSection?.rules) {
const hasTrikayaRef = guidelinesSection.rules.some(rule =>
rule.trikaya_ebene || (typeof rule === 'string' && rule.includes('Trikāya'))
);
if (!hasTrikayaRef) {
suggestions.push({
path,
issue: "Fehlende Trikāya-Referenz in den Guidelines",
fix: `Füge eine Regel hinzu wie:
- "Jeder Kernpunkt MUSS einer Trikāya-Ebene (Dharmakāya/Sambhogakāya/Nirmāṇakāya) zugeordnet sein."`,
line: JSON.stringify(sigillin, null, 2).split('\n').findIndex(l => l.includes('"guidelines"'))
});
}
}

// 2. Prüfe CREP-Bezüge in Kernpunkten
const kernpunkteSection = sigillin.sigillin.content.sections.find(s => s.id === 'crep_explained');
if (kernpunkteSection?.table) {
const crepMetrics = ['Coherence', 'Resonance', 'Emergence', 'Poetics'];
const coveredMetrics = kernpunkteSection.table.map((row: any) => row[0]);
const missingMetrics = crepMetrics.filter(m => !coveredMetrics.includes(m));
if (missingMetrics.length > 0) {
suggestions.push({
path,
issue: `Fehlende CREP-Metriken: ${missingMetrics.join(', ')}`,
fix: `Ergänze Zeilen für ${missingMetrics.join(' und ')} in der CREP-Tabelle. Beispiel:
- ["${missingMetrics[0]}", "Kurzbeschreibung", "Beispiel"]`,
line: JSON.stringify(sigillin, null, 2).split('\n').findIndex(l => l.includes('"crep_explained"'))
});
}
}

// 3. Prüfe "Nächste Schritte"
const nextStepsSection = sigillin.sigillin.content.sections.find(s => s.id === 'naechste_schritte');
if (!nextStepsSection) {
suggestions.push({
path,
issue: "Fehlender Abschnitt 'Nächste Schritte'",
fix: `Füge einen Abschnitt hinzu:
{
"id": "naechste_schritte",
"title": "Nächste Schritte",
"examples": [
{
"handlung": "Mini-Sigillin für [Ziel] anlegen",
"messpunkt": "CREP-Score nach 1 Woche (Ziel: C > 0.85)"
}
]
}`,
line: sigillin.sigillin.content.sections.length * 2 // Approximation
});
} else if (nextStepsSection.examples?.length === 0) {
suggestions.push({
path,
issue: "Leere 'Nächste Schritte'-Beispiele",
fix: `Füge mindestens ein Beispiel hinzu:
{
"handlung": "Mini-Sigillin für [konkretes Ziel] anlegen",
"messpunkt": "CREP-Score nach [Zeitraum] (Ziel: [Wert])"
}`,
line: JSON.stringify(sigillin, null, 2).split('\n').findIndex(l => l.includes('"naechste_schritte"'))
});
}
}

if (suggestions.length === 0) {
console.log('✅ Alle Sigillins sind korrekt!');
return;
}

console.log(`🔧 ${suggestions.length} Korrekturvorschläge gefunden:`);
suggestions.forEach(s => {
console.log(`\n📄 ${s.path}`);
console.log(`🚨 ${s.issue}`);
console.log(`🔧 Vorschlag: ${s.fix}`);
if (s.line) console.log(` (ca. Zeile ${s.line})`);
});

// Interaktive Abfrage: Soll automatisch korrigiert werden?
const { confirm } = await prompt({
type: 'confirm',
name: 'confirm',
message: `Sollen die ${suggestions.length} Vorschläge automatisch angewendet werden?`,
initial: false
});

if (confirm) {
suggestions.forEach(s => {
const raw = readFileSync(s.path, 'utf-8');
const sigillin = JSON.parse(raw);

if (s.issue.includes('Trikāya-Referenz')) {
sigillin.sigillin.content.sections
.find((s: any) => s.id === 'guidelines')
?.rules.push("Jeder Kernpunkt MUSS einer Trikāya-Ebene (Dharmakāya/Sambhogakāya/Nirmāṇakāya) zugeordnet sein.");
}
else if (s.issue.includes('CREP-Metriken')) {
const missingMetrics = s.fix.match(/Ergänze Zeilen für (.*?) in der CREP-Tabelle/)?.[1].split(' und ');
if (missingMetrics) {
sigillin.sigillin.content.sections
.find((s: any) => s.id === 'crep_explained')
?.table.push(
...missingMetrics.map(metric => [metric, "Kurzbeschreibung", "Beispiel"])
);
}
}
else if (s.issue.includes('Nächste Schritte')) {
if (!sigillin.sigillin.content.sections.some((s: any) => s.id === 'naechste_schritte')) {
sigillin.sigillin.content.sections.push({
id: 'naechste_schritte',
title: 'Nächste Schritte',
examples: [{
handlung: "Mini-Sigillin für [Ziel] anlegen",
messpunkt: "CREP-Score nach 1 Woche (Ziel: C > 0.85)"
}]
});
} else if (sigillin.sigillin.content.sections
.find((s: any) => s.id === 'naechste_schritte')
?.examples?.length === 0) {
sigillin.sigillin.content.sections
.find((s: any) => s.id === 'naechste_schritte')
?.examples.push({
handlung: "Mini-Sigillin für [konkretes Ziel] anlegen",
messpunkt: "CREP-Score nach [Zeitraum] (Ziel: [Wert])"
});
}
}

writeFileSync(s.path, JSON.stringify(sigillin, null, 2));
console.log(`✅ Korrigiert: ${s.path}`);
});
}
}

analyzeAndFixSigillins().catch(console.error);
```

function calculateDynamicCREP(sigillin: Sigillin): { score: number; issues: string[] } {
const issues: string[] = [];
let score = 1.0;

// 1. Trikāya-Abdeckung
const hasTrikaya = sigillin.sigillin.content.sections.some(s =>
s.id === 'guidelines' && s.rules?.some((r: any) => r.includes('Trikāya'))
);
if (!hasTrikaya) {
score -= 0.1;
issues.push("Fehlende Trikāya-Referenz");
}

// 2. CREP-Vollständigkeit
const crepSection = sigillin.sigillin.content.sections.find(s => s.id === 'crep_explained');
if (!crepSection?.table || crepSection.table.length < 4) {
score -= 0.2;
issues.push("Unvollständige CREP-Tabelle");
}

// 3. Nächste Schritte
const nextSteps = sigillin.sigillin.content.sections.find(s => s.id === 'naechste_schritte');
if (!nextSteps?.examples || nextSteps.examples.length === 0) {

### 1\. `pre-commit`-Skript zur Validierung geänderter Sigillins

Dieses Skript ist der erste Schritt zur **Qualitätssicherung**. Es erhöht die `Kohärenz`, indem es sicherstellt, dass nur gültige `Sigillin`-Dateien in das Repository aufgenommen werden. Es scannt nur die Dateien, die du mit `git add` hinzugefügt hast, was es extrem schnell macht.

```bash
#!/bin/sh
# .githooks/pre-commit
#
# Überprüft, ob geänderte Sigillin-Dateien valide sind,
# bevor sie ins Repo committet werden.
#
echo "🔎 Validating changed Sigillin files..."
# Finde alle hinzugefügten oder geänderten .json, .yaml und .md Dateien im "sigils" oder "docs/sigillin" Verzeichnis
STAGED_FILES=$(git diff --cached --name-only --diff-filter=ACM | grep -E '^sigils/.*.(json|yaml)$|^docs/sigillin/.*.md$')
if [ -z "$STAGED_FILES" ]; then
echo "✅ No Sigillin files to validate. Skipping."
exit 0
fi
# Führe den Validator für jede gefundene Datei aus
for file in $STAGED_FILES; do
echo "Running validation on: $file"
# Nutze pnpm, um den Validator auszuführen.
# Dies setzt voraus, dass der Pfad zum Validator in deiner package.json hinterlegt ist.
# Passe den Pfad an, wenn nötig.
if ! pnpm validate:sigillins "$file"; then
echo "❌ Validation failed for $file. Aborting commit."
exit 1
fi
done
echo "✅ All staged Sigillin files are valid."
exit 0
```

**Anwendung:** Lege das Skript im `.githooks`-Verzeichnis ab und mache es ausführbar (`chmod +x .githooks/pre-commit`). Du kannst auch eine Bibliothek wie `husky` verwenden, um dies zu automatisieren.

### 2\. `JUnit XML`-Bericht für CI/CD und Governance

Dieser Teil sorgt für die **Automatisierung der Berichterstattung**. Er erzeugt einen standardisierten `JUnit XML`-Bericht, der von CI/CD-Systemen wie GitHub Actions, Jenkins oder GitLab CI verstanden wird. Dies ist ein entscheidendes `Nirmāṇakāya`-Artefakt für die **transparente Governance** des Projekts.

```typescript
// scripts/generate-junit-report.ts
import { writeFileSync } from 'fs';
import { XMLBuilder } from 'fast-xml-parser';
import { runValidation } from './sigillin-validator'; // Angenommen, diese Funktion existiert
import path from 'path';
// Beispiel-Daten (in realer Anwendung aus runValidation() beziehen)
async function generateJUnitReport(filePaths: string[]): Promise<void> {
const testSuites = [];
let totalTests = 0;
let totalFailures = 0;
for (const filePath of filePaths) {
const result = await runValidation(filePath); // `runValidation` aus dem letzten Entwurf
const fileName = path.basename(filePath);
const testCase = {
'@_name': `Validation of ${fileName}`,
'@_classname': `Sigillin.${path.dirname(filePath).split(path.sep).pop()}`,
'failure': result.valid ? undefined : {
'@_message': 'Sigillin validation failed.',
'#text': result.errors?.join('\n') || 'No errors reported.'
}
};
if (!result.valid) {
totalFailures++;
}
totalTests++;
testSuites.push({
'@_name': `SigillinTestSuite.${fileName}`,
'@_tests': 1,
'@_failures': result.valid ? 0 : 1,
'testcase': testCase
});
}
const junitObject = {
'testsuites': {
'@_tests': totalTests,
'@_failures': totalFailures,
'testsuite': testSuites
}
};
const builder = new XMLBuilder({
ignoreAttributes: false,
format: true
});
const xmlContent = builder.build(junitObject);
const outputPath = 'reports/sigillin-junit-report.xml';
writeFileSync(outputPath, xmlContent);
console.log(`✅ JUnit report generated at ${outputPath}`);
}
// Beispiel-Nutzung
const allSigillinFiles = [
'sigils/bridges/chatgpt/chatgpt-bridge.sigil.yaml',
'sigils/bridges/gemini/gemini-bridge.sigil.yaml'
// ... alle anderen YAML/JSON/MD-Dateien, die geprüft werden sollen
];
generateJUnitReport(allSigillinFiles);
```

### 3\. `Markdown Summary` für Governance-Reports

Dieser Bericht ist für die `Dharmakāya`-Ebene gedacht: Er macht die Qualität eures Wissensmanagements **für menschliche Leser sichtbar**. Er fasst die Validierungsergebnisse in einem klaren, präzisen `Markdown`-Format zusammen, das leicht in Governance-Dokumente oder PR-Beschreibungen integriert werden kann. Dies erhöht die `Poetik` und `Resonanz` eurer internen Kommunikation.

```typescript
// scripts/generate-markdown-report.ts
import { writeFileSync } from 'fs';
import { runValidation } from './sigillin-validator'; // Angenommen, diese Funktion existiert
import path from 'path';
async function generateMarkdownReport(filePaths: string[]): Promise<void> {
let markdownContent = `# 🛡️ Sigillin Validation Summary\n\n`;
markdownContent += `* **Generated on:** ${new Date().toUTCString()}\n\n`;
markdownContent += `| File | Status | CREP Score | Trikāya Coverage |\n`;
markdownContent += `|---|---|---|---|\n`;
let allValid = true;
for (const filePath of filePaths) {
const result = await runValidation(filePath);
const fileName = path.basename(filePath);
const status = result.valid ? '✅ Pass' : '❌ Fail';
const crepScore = result.crepScore ? `${(result.crepScore * 100).toFixed(1)}%` : 'N/A';
const trikayaCoverage = result.qualityAssessment?.hasTrikayaReference ? '✅ Yes' : '❌ No';
markdownContent += `| ${fileName} | ${status} | ${crepScore} | ${trikayaCoverage} |\n`;
if (!result.valid) {
allValid = false;
markdownContent += `\n**Errors for ${fileName}:**\n`;
markdownContent += result.errors?.map(err => `* ${err}`).join('\n') + '\n\n';
}
}
if (allValid) {
markdownContent += `\n**✅ All Sigillins are valid. Fraktale Kohärenz ist gewährleistet.**\n`;
} else {
markdownContent += `\n**🚨 Warning: Some Sigillins failed validation. Fraktale Kohärenz ist gebrochen.**\n`;
}
const outputPath = 'reports/sigillin-report.md';
writeFileSync(outputPath, markdownContent);
console.log(`✅ Markdown report generated at ${outputPath}`);
}
// Beispiel-Nutzung
const allSigillinFiles = [
'sigils/bridges/chatgpt/chatgpt-bridge.sigil.yaml',
'sigils/bridges/gemini/gemini-bridge.sigil.yaml',
'docs/sigillin/bridges/CHATGPT_SIGILLIN.md'
];
generateMarkdownReport(allSigillinFiles);
```

*(Ziel: Bridges sind **nicht nur formal**, sondern tragen CREP + Trikāya + „Next Action“ – eure eigene Doktrin als Maschine lesbare Pflicht.)*&#x20;

**B) Validator-Hook** `scripts/validate-sigillins.mjs` – *post-Schema Checks*:

```js
// Pseudocode-Ausschnitt, am Ende der Schema-Validierung:
const hasCREP = (o) => {
const c = o.crep || {};
const count = ["coherence","resonance","emergence","poetics"].filter(k => typeof c[k] === "number").length;
return count >= 2;
};
const hasNextAction = (o) => Boolean(o.guidelines && typeof o.guidelines.next_action === "string" && o.guidelines.next_action.length >= 12);
const hasTrikaya = (o) => Array.isArray(o.trikaya) && o.trikaya.length >= 1;

for (const sig of allSigillins) {
const fails = [];
if (!hasCREP(sig)) fails.push("CREP: need ≥2 metrics");
if (!hasTrikaya(sig)) fails.push("Trikāya: need ≥1 layer");
if (!hasNextAction(sig)) fails.push("Guidelines.next_action required");
if (fails.length) {
report(sig.id || sig.title, fails);
hardFail = true;
}
}
process.exit(hardFail ? 1 : 0);
```
